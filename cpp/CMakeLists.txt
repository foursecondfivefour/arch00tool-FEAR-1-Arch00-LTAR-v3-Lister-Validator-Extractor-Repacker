cmake_minimum_required(VERSION 3.16)
project(arch00tool_cpp LANGUAGES CXX)
include(FetchContent)
# Prefer a system development package; otherwise obtain a pinned zlib release.
find_package(ZLIB QUIET)
if(NOT ZLIB_FOUND)
  set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(zlib URL https://github.com/madler/zlib/archive/refs/tags/v1.3.1.tar.gz
    URL_HASH SHA256=17e88863f3600672ab49182f217281b6fc4d3c762bde361935e436a95214d05c)
  FetchContent_MakeAvailable(zlib)
  set(ARCH00_ZLIB zlibstatic)
else()
  set(ARCH00_ZLIB ZLIB::ZLIB)
endif()
add_library(arch00_core src/arch00.cpp src/payload.cpp src/repack.cpp)
target_link_libraries(arch00_core PRIVATE ${ARCH00_ZLIB})
target_include_directories(arch00_core PUBLIC include)
target_compile_features(arch00_core PUBLIC cxx_std_17)
add_executable(arch00tool-cpp src/main.cpp)
target_link_libraries(arch00tool-cpp PRIVATE arch00_core)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME parser COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_parser.py" "$<TARGET_FILE:arch00tool-cpp>")
    add_test(NAME cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py" "$<TARGET_FILE:arch00tool-cpp>")
    add_test(NAME repack COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_repack.py" "$<TARGET_FILE:arch00tool-cpp>")
    add_test(NAME payload COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_payload.py" "$<TARGET_FILE:arch00tool-cpp>")
  endif()
endif()
