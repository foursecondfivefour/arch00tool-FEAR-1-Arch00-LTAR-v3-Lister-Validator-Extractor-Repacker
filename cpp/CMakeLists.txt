cmake_minimum_required(VERSION 3.16)
project(arch00tool_cpp LANGUAGES CXX)
include(FetchContent)
# Prefer a system development package; otherwise obtain a pinned zlib release.
find_package(ZLIB QUIET)
if(NOT ZLIB_FOUND)
  set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(zlib URL https://github.com/madler/zlib/archive/refs/tags/v1.3.1.tar.gz
    URL_HASH SHA256=17e88863f3600672ab49182f217281b6fc4d3c762bde361935e436a95214d05c)
  FetchContent_MakeAvailable(zlib)
  set(ARCH00_ZLIB zlibstatic)
else()
  set(ARCH00_ZLIB ZLIB::ZLIB)
endif()
find_package(Threads REQUIRED)
add_library(arch00_core src/arch00.cpp src/payload.cpp src/repack.cpp)
target_link_libraries(arch00_core PRIVATE ${ARCH00_ZLIB} Threads::Threads)
target_include_directories(arch00_core PUBLIC include)
target_compile_features(arch00_core PUBLIC cxx_std_17)
add_executable(arch00tool-cpp src/main.cpp)
target_link_libraries(arch00tool-cpp PRIVATE arch00_core)
if(MINGW)
  target_link_options(arch00tool-cpp PRIVATE -municode)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME benchmark-smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/benchmarks/benchmark.py" --cpp "$<TARGET_FILE:arch00tool-cpp>" --files 4 --size-kib 16 --repeat 1 --jobs 2)
    add_test(NAME parallel COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_parallel.py" "$<TARGET_FILE:arch00tool-cpp>")
    add_test(NAME unicode-paths COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_unicode_paths.py" "$<TARGET_FILE:arch00tool-cpp>")
    add_test(NAME parser COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_parser.py" "$<TARGET_FILE:arch00tool-cpp>")
    add_test(NAME cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py" "$<TARGET_FILE:arch00tool-cpp>")
    add_test(NAME repack COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_repack.py" "$<TARGET_FILE:arch00tool-cpp>")
    add_test(NAME payload COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_payload.py" "$<TARGET_FILE:arch00tool-cpp>")
  endif()
endif()

option(ARCH00_BUILD_GUI "Build the experimental browser GUI" OFF)
if(ARCH00_BUILD_GUI)
  set(HTTPLIB_REQUIRE_OPENSSL OFF CACHE BOOL "" FORCE)
  set(HTTPLIB_TEST OFF CACHE BOOL "" FORCE)
  set(JSON_BuildTests OFF CACHE INTERNAL "")
  FetchContent_Declare(httplib GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git GIT_TAG a7bc00e3307fecdb4d67545e93be7b88cfb1e186)
  FetchContent_Declare(nlohmann_json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03)
  FetchContent_MakeAvailable(httplib nlohmann_json)
  add_executable(arch00tool-gui src/gui.cpp)
  target_link_libraries(arch00tool-gui PRIVATE arch00_core httplib::httplib nlohmann_json::nlohmann_json)
  if(BUILD_TESTING AND Python3_Interpreter_FOUND)
    add_test(NAME gui COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_gui.py" "$<TARGET_FILE:arch00tool-gui>")
  endif()
endif()
